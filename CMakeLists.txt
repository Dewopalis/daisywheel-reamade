cmake_minimum_required(VERSION 3.20)
project(Daisywheel VERSION 0.5.1 LANGUAGES C CXX)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
find_package(SDL2 REQUIRED CONFIG)
add_executable(daisywheel src/main.cpp src/overlaywidgets.h src/settingsnavigation.cpp src/settingsnavigation.h src/soundeffects.cpp src/soundeffects.h)
target_link_libraries(daisywheel PRIVATE Qt6::Widgets SDL2::SDL2)
if(APPLE)
  set_target_properties(daisywheel PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER org.daisywheel.reamade
    MACOSX_BUNDLE_BUNDLE_NAME "Daisywheel REAmade"
  )
endif()
if(UNIX AND NOT APPLE)
  target_sources(daisywheel PRIVATE src/appimageinstaller.cpp src/portalpointer.cpp src/portalpointer.h)
  find_package(Qt6 REQUIRED COMPONENTS WaylandClient DBus)
  find_package(LayerShellQt 6.3 REQUIRED)
  if(LayerShellQt_VERSION VERSION_LESS 6.6)
    target_compile_definitions(daisywheel PRIVATE DAISYWHEEL_LEGACY_LAYER_SCREEN)
  else()
    target_compile_definitions(daisywheel PRIVATE DAISYWHEEL_LAYER_SHELL_MODERN_API)
  endif()
  qt_generate_wayland_protocol_client_sources(daisywheel PRIVATE_CODE FILES ${CMAKE_CURRENT_SOURCE_DIR}/protocols/input-method-unstable-v1.xml)
  target_sources(daisywheel PRIVATE src/kdeinput.cpp)
  target_link_libraries(daisywheel PRIVATE Qt6::WaylandClient Qt6::DBus LayerShellQt::Interface)
  target_compile_definitions(daisywheel PRIVATE DAISYWHEEL_KDE)
  target_compile_options(daisywheel PRIVATE -include wayland-client-protocol.h)
endif()
install(TARGETS daisywheel
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  BUNDLE DESTINATION .
)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/daisywheel-reamade)
if(UNIX AND NOT APPLE)
  install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/daisywheel-reamade
          RENAME copyright)
  install(FILES packaging/org.daisywheel.keyboard.desktop
          DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/daisywheel-reamade.svg
          DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
include(packaging/CPackConfig.cmake)
if(WIN32)
  target_sources(daisywheel PRIVATE src/windowsfocus.cpp)
  target_compile_definitions(daisywheel PRIVATE NOMINMAX SDL_MAIN_HANDLED)
  target_link_libraries(daisywheel PRIVATE user32 ole32 oleaut32 uuid)
endif()
enable_testing()
add_executable(wheel-tests tests/wheel.cpp)
target_include_directories(wheel-tests PRIVATE src)
add_test(NAME wheel-selection COMMAND wheel-tests)
add_executable(placement-tests tests/placement.cpp)
target_include_directories(placement-tests PRIVATE src)
target_link_libraries(placement-tests PRIVATE Qt6::Core)
add_test(NAME overlay-placement COMMAND placement-tests)
add_executable(settings-navigation-tests tests/settingsnavigation.cpp src/settingsnavigation.cpp src/settingsnavigation.h)
target_include_directories(settings-navigation-tests PRIVATE src)
target_link_libraries(settings-navigation-tests PRIVATE Qt6::Widgets)
add_test(NAME settings-navigation COMMAND settings-navigation-tests)
set_tests_properties(settings-navigation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(pointer-button-state-tests tests/pointerbuttonstate.cpp)
target_include_directories(pointer-button-state-tests PRIVATE src)
add_test(NAME pointer-button-state COMMAND pointer-button-state-tests)
if(WIN32)
  add_executable(windows-input-tests tests/windows.cpp)
  target_include_directories(windows-input-tests PRIVATE src)
  target_compile_definitions(windows-input-tests PRIVATE NOMINMAX)
  target_link_libraries(windows-input-tests PRIVATE Qt6::Widgets user32)
  add_executable(windows-focus-tests tests/windowsfocus.cpp src/windowsfocus.cpp)
  target_include_directories(windows-focus-tests PRIVATE src)
  target_compile_definitions(windows-focus-tests PRIVATE NOMINMAX)
  target_link_libraries(windows-focus-tests PRIVATE Qt6::Widgets user32 ole32 oleaut32 uuid)
  # Requires an interactive desktop; run explicitly rather than during headless CTest.
endif()
